108/*==================================================================================================
109* RX PADDING VALIDATION CONFIG
110==================================================================================================*/
111/* Reject (drop) received SF / CF frames whose padding bytes deviate from
112 * CANTP_PADDING_BYTE_VALUE. Default STD_OFF: the frame is still processed and
113 * only a CANTP_E_RX_UNEXP_PADDING runtime error is reported (ISO 15765-2
114 * tolerance towards senders that do not pad). FC frames are always exempt
115 * because their padding is sender-specific. FF frames carry no padding area
116 * on classic CAN (2 PCI + 6 data bytes fill the frame exactly). */
117#ifndef CANTP_REJECT_INVALID_PADDING
118#define CANTP_REJECT_INVALID_PADDING (STD_OFF)
119#endif
