46/* LinTp channel state type */
47typedef enum
48{
49 LINTP_CH_IDLE = 0, /* Channel is idle */
50 LINTP_CH_BUSY, /* Channel is busy processing a message */
51 LINTP_CH_TX_ACTIVE, /* Channel is transmitting */
52 LINTP_CH_RX_ACTIVE /* Channel is receiving */
53} LinTp_ChannelStateType;
54
55/* LinTp internal state type */
56typedef enum
57{
58 LINTP_STATE_UNINIT = 0, /* Module not initialized */
59 LINTP_STATE_INIT /* Module initialized */
60} LinTp_StateType;

62/* LinTp transmission state machine */
63typedef enum
64{
65 LINTP_TX_IDLE = 0, /* No transmission active */
66 LINTP_TX_SF, /* Transmitting Single Frame */
67 LINTP_TX_FF, /* Transmitting First Frame */
68 LINTP_TX_CF, /* Transmitting Consecutive Frame */
69 LINTP_TX_WAIT_FC, /* Waiting for Flow Control */
70 LINTP_TX_WAIT_TX_CONFIRM, /* Waiting for transmission confirmation */
71 LINTP_TX_COMPLETED, /* Transmission completed */
72 LINTP_TX_ERROR /* Transmission error */
73} LinTp_TxStateType;
74
75/* LinTp reception state machine */
76typedef enum
77{
78 LINTP_RX_IDLE = 0, /* No reception active */
79 LINTP_RX_WAIT_FF, /* Waiting for First Frame */
80 LINTP_RX_WAIT_CF, /* Waiting for Consecutive Frame */
81 LINTP_RX_SEND_FC, /* Sending Flow Control */
82 LINTP_RX_COMPLETED, /* Reception completed */
83 LINTP_RX_ERROR /* Reception error */
84} LinTp_RxStateType;
85
86/* PCI (Protocol Control Information) types */
87typedef enum
88{
89 LINTP_PCI_SF = 0x00, /* Single Frame */
90 LINTP_PCI_FF = 0x10, /* First Frame */
91 LINTP_PCI_CF = 0x20, /* Consecutive Frame */
92 LINTP_PCI_FC = 0x30 /* Flow Control */
93} LinTp_PciType;
94
95/* Flow Control status */
96typedef enum
97{
98 LINTP_FC_CTS = 0x00, /* Continue To Send */
99 LINTP_FC_WAIT = 0x01, /* Wait */
100 LINTP_FC_OVFLW = 0x02 /* Overflow/Abort */
101} LinTp_FlowStatusType;
102
103/* Parameter change type */
104typedef enum
105{
106 LINTP_PARAMETER_BS = 0x01, /* Block Size */
107 LINTP_PARAMETER_STMIN = 0x02 /* Separation Time Minimum */
108} LinTp_ParameterType;

111typedef struct
112{
113 uint16 ChannelId; /* LinTp channel ID */
114 uint16 LinIfChannelId; /* Associated LinIf channel ID */
115 uint32 N_As; /* Sender response timeout (ms) */
116 uint32 N_Cs; /* Sender confirmation timeout (ms) */
117 uint32 N_Cr; /* Receiver confirmation timeout (ms) */
118 uint8 DefaultBs; /* Default Block Size */
119 uint8 DefaultStMin; /* Default STmin (ms) */
120 boolean TransmitCancellation; /* Transmit cancellation supported */
121} LinTp_ChannelConfigType;

123/* NSC (Network Service Connection) configuration */
124typedef struct
125{
126 uint16 NsduId; /* N-SDU identifier */

130 boolean IsTx; /* TRUE = Tx connection, FALSE = Rx connection */
131} LinTp_NsduConfigType;

133/* Channel runtime data */

136 LinTp_ChannelStateType State; /* Channel state */
137 LinTp_TxStateType TxState; /* Transmission sub-state */

139 uint16 CurrentNsduId; /* Current N-SDU being processed */
140 PduLengthType DataLength; /* Total data length */
141 PduLengthType DataIndex; /* Current data index */
142 PduLengthType BufferSize; /* Available buffer size */
143 uint8 SequenceNumber; /* Consecutive frame sequence number */
144 uint8 BlockSize; /* Current block size */
145 uint8 BlockCount; /* CF frames sent in current block */
146 uint8 StMin; /* Separation time minimum */
147 uint8 WftCount; /* Wait frame transmission count */
148 uint32 TimeoutCounter; /* Timeout counter */
149 boolean BufferProvided; /* Buffer provided by PduR */
150 uint8 *TxBuffer; /* Transmit buffer pointer */
151 uint8 *RxBuffer; /* Receive buffer pointer */
152} LinTp_ChannelRuntimeType;
153

158 const LinTp_NsduConfigType *NsduConfig; /* N-SDU configurations */
159 uint8 ChannelCount; /* Number of channels */
160 uint8 NsduCount; /* Number of N-SDUs */
161} LinTp_ConfigType;
