111/*==================================================================================================
112* CANIF PARTIAL NETWORKING / RX DISPATCH / TX QUEUE CONFIG
113==================================================================================================*/
114/* Compile switch: use the Hoh-bucketed Rx dispatch lookup table built by
115 * CanIf_Init (O(bucket) matching) instead of the linear scan over all Rx
116 * L-PDUs. When disabled, or for hardware object handles outside the table
117 * range, the linear scan is used as fallback. */
118#ifndef CANIF_USE_RX_LOOKUP_TABLE
119#define CANIF_USE_RX_LOOKUP_TABLE (STD_ON)
120#endif
121/* Highest hardware object handle covered by the Rx dispatch lookup table */
122#ifndef CANIF_RX_LOOKUP_MAX_HOH
123#define CANIF_RX_LOOKUP_MAX_HOH (16U)
124#endif
125/* Depth of the Tx retry queue buffering frames that Can_Write rejected with
126 * CAN_BUSY; drained by CanIf_TxQueueMainFunction. */
127#ifndef CANIF_TX_QUEUE_DEPTH
128#define CANIF_TX_QUEUE_DEPTH (16U)
129#endif
130/* Maximum payload length bufferable in one Tx queue entry (classic CAN) */
131#ifndef CANIF_TX_QUEUE_DATA_LENGTH
132#define CANIF_TX_QUEUE_DATA_LENGTH (8U)
133#endif
