458/*==================================================================================================
459 * SECOC COMPATIBILITY DEFINITIONS
460 *================================================================================================*/
461/* NOTE (P1 Phase 8): these job IDs must resolve to the MAC jobs of the
462 * default job table in Csm.c (CSM_JOB_ID_MAC_GENERATE_DEFAULT = 4,
463 * CSM_JOB_ID_MAC_VERIFY_DEFAULT = 5). The previous values 1/2 resolved to
464 * the HASH/ENCRYPT jobs, so every SecOC MAC request failed the Csm
465 * key-usage check and the whole SecOC crypto chain was dead. */
466#ifndef CSM_JOB_ID_MAC_GENERATE_1
467#define CSM_JOB_ID_MAC_GENERATE_1 (4U)
468#endif
469#ifndef CSM_JOB_ID_MAC_VERIFY_1
470#define CSM_JOB_ID_MAC_VERIFY_1 (5U)
471#endif
472/* NOTE (P1 Phase 8): Csm.c decodes the mode parameter as bit flags
473 * (START = 0x01, UPDATE = 0x02, FINISH = 0x04). STREAMSTART (0x01) never
474 * sets the FINISH flag, so MAC generation/verification never produced a
475 * result. SINGLESHOT (0x05 = START|FINISH) is the mode SecOC must use. */
476#ifndef CSM_OPERATIONMODE_STREAMSTART
477#define CSM_OPERATIONMODE_STREAMSTART (0x01U)
478#endif
479#ifndef CSM_OPERATIONMODE_SINGLESHOT
480#define CSM_OPERATIONMODE_SINGLESHOT (0x05U)
481#endif
