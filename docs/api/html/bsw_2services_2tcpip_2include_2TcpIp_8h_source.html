255/* VLAN configuration (B1): adaption-layer VLAN settings for the
256 * single Ethernet interface. Actual on-wire tag insertion is performed
257 * by lwIP (LWIP_VLAN_PCP, per-PCB TCI via netif hints) or by the
258 * Ethernet Switch (EthSwt, see B2); this module validates and tracks
259 * the interface VLAN membership. */
260typedef struct {
261 boolean VlanEnabled; /* Tag outgoing frames with the VLAN id */
262 uint16 VlanId; /* 12-bit VID (0..4095); 0 = untagged */
263 uint8 VlanPriority; /* 802.1p PCP (0..7) */
264 boolean DropUntagged; /* Drop untagged ingress frames */
265} TcpIp_VlanConfigType;

267/* Statistics (B1): monotonic counters aligned with AUTOSAR
268 * TcpIp_GetStatistics-class capability. Reset via TcpIp_ResetStatistics. */
269typedef struct {
270 uint32 TxPackets; /* datagrams/segments handed to the stack */
271 uint32 TxBytes; /* payload bytes transmitted */
272 uint32 RxPackets; /* segments/datagrams queued by RxIndication */
273 uint32 RxBytes; /* payload bytes received */
274 uint32 TxErrors; /* transmit failures */
275 uint32 RxErrors; /* receive errors (invalid socket/input) */
276 uint32 RxOverflows; /* RX queue / chunk overflow */
277 uint32 TcpActiveOpens; /* TCP connect() initiations */
278 uint32 TcpPassiveOpens; /* incoming SYNs accepted on listeners */
279 uint32 TcpEstablishedCount;/* transitions into ESTABLISHED */
280 uint32 TcpCloseCount; /* TCP connections closed (slot freed) */
281 uint32 SocketCreateCount; /* TcpIp_Create successes */
282 uint32 SocketCloseCount; /* slots released (Close/Abort/pending) */
283} TcpIp_StatisticsType;
