366#define CRYPTO_STACK_MEMCPY(dst, src, size) \
367 do { \
368 if ((dst) != NULL_PTR && (src) != NULL_PTR && (size) > 0U) { \
369 for (uint32 i = 0U; i < (size); i++) { \
370 ((uint8*)(dst))[i] = ((const uint8*)(src))[i]; \
371 } \
372 } \
373 } while(0)
