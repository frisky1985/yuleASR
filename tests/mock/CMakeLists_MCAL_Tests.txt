#----------------------------------------------------------------------
# Helper function: add_mcal_test
#
# Usage:
#   add_mcal_test(
#     NAME        <target_name>
#     TEST_FILE   <path/to/test.c>
#     MCAL_SRCS   <src1.c> [src2.c ...]   # Real production MCAL sources
#     MOCK_SRCS   <mock1.c> [mock2.c ...] # Mock/stub sources
#     DEFINES     <DEF1=VAL1> [DEF2=VAL2] # Compile definitions
#   )
#----------------------------------------------------------------------
function(add_mcal_test)
    cmake_parse_arguments(ARG "" "NAME;TEST_FILE" "MCAL_SRCS;MOCK_SRCS;DEFINES" ${ARGN})

    set(TEST_TARGET "${ARG_NAME}")

    # Collect sources
    set(SOURCES
        ${ARG_TEST_FILE}
        ${UNITY_C}
        ${ARG_MCAL_SRCS}
        ${ARG_MOCK_SRCS}
    )

    # Add MockHAL library
    add_executable(${TEST_TARGET} ${SOURCES})

    # Force-include mock_hal.h before all translation units so production
    # source files (which use REG_READ32/REG_WRITE32) have the mock
    # function declarations available.
    target_compile_options(${TEST_TARGET} PRIVATE
        -include ${MOCK_DIR}/mock_hal.h
    )

    # Include directories
    target_include_directories(${TEST_TARGET} PRIVATE
        ${MCAL_TEST_INCLUDES}
        ${CMAKE_CURRENT_BINARY_DIR}       # For generated config headers
    )

    # Compile definitions — redirect REG_READ32/REG_WRITE32 to mock_hal
    target_compile_definitions(${TEST_TARGET} PRIVATE
        # Redirect register access macros to MockHAL
        REG_READ32=mock_hal_read32
        REG_WRITE32=mock_hal_write32
        REG_READ16=mock_hal_read16
        REG_WRITE16=mock_hal_write16
        REG_READ8=mock_hal_read8
        REG_WRITE8=mock_hal_write8
        ${ARG_DEFINES}
    )

    # Compile options
    target_compile_options(${TEST_TARGET} PRIVATE
        -Wall
        -Wextra
        -Wno-unused-parameter
        -Wno-unused-variable
        -Wno-unused-function
        -Wno-int-conversion
        -Wno-implicit-int
        -Wno-implicit-function-declaration
        -Wno-excess-initializers
        -g
        -O0
    )

    # Link MockHAL
    target_link_libraries(${TEST_TARGET} PRIVATE MockHAL)

    # Link coverage when enabled
    if(ENABLE_COVERAGE)
        target_compile_options(${TEST_TARGET} PRIVATE --coverage)
        target_link_libraries(${TEST_TARGET} PRIVATE gcov)
    endif()

    # Register test
    add_test(NAME ${TEST_TARGET} COMMAND ${TEST_TARGET})

    message(STATUS "  MCAL test: ${TEST_TARGET}")
endfunction()
